# Kimi 抽帧视频解析 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将第一阶段视频识别从“整段视频 base64 直传”改为“浏览器本地抽帧后发送代表帧”，并在低置信度时自动补抽重试一次。

**Architecture:** 保留现有两阶段链路和状态流，在浏览器端新增抽帧与代表帧筛选能力，并将第一阶段请求改造成多模态图片输入。第一轮识别结果若缺少核心景点或置信度不高，则再补抽一轮新帧重试，第二阶段景区补全逻辑保持不变。

**Tech Stack:** React 18, TypeScript, Vite, Zustand, Vitest, Canvas/HTMLVideoElement

export type SampledFrame = {
  dataUrl: string
  sharpness: number
  brightness: number
  hash: string
  ratio: number
}

const SAMPLE_COUNT = 10
const SAMPLE_START = 0.1
const SAMPLE_END = 0.9

export function buildSampleRatios(): number[] {
  const step = (SAMPLE_END - SAMPLE_START) / (SAMPLE_COUNT - 1)
  return Array.from({ length: SAMPLE_COUNT }, (_, index) =>
    Number((SAMPLE_START + step * index).toFixed(4)),
  )
}

export function buildRetrySampleRatios(primaryRatios: number[]): number[] {
  return [0, 2, 4, 6].map((index) =>
    Number((((primaryRatios[index] ?? 0) + (primaryRatios[index + 1] ?? 0)) / 2).toFixed(4)),
  )
}

function hammingDistance(left: string, right: string): number {
  return left.split("").reduce((count, bit, index) => count + Number(bit !== right[index]), 0)
}

export function pickRepresentativeFrames(frames: SampledFrame[], limit: number): SampledFrame[] {
  const picked: SampledFrame[] = []

  for (const frame of [...frames].sort((a, b) => b.sharpness - a.sharpness)) {
    const isNearDuplicate = picked.some((item) => hammingDistance(item.hash, frame.hash) <= 1)
    if (!isNearDuplicate) picked.push(frame)
    if (picked.length === limit) break
  }

  return picked.sort((a, b) => a.ratio - b.ratio)
}

```ts
export async function extractVideoKeyframes(file: File, ratios: number[]): Promise<SampledFrame[]> {
  const objectUrl = URL.createObjectURL(file)
  const video = document.createElement("video")
  video.preload = "auto"
  video.muted = true
  video.src = objectUrl

  await new Promise<void>((resolve, reject) => {
    video.onloadedmetadata = () => resolve()
    video.onerror = () => reject(new Error("VIDEO_METADATA_LOAD_FAILED"))
  })

  const canvas = document.createElement("canvas")
  const context = canvas.getContext("2d")
  if (!context) throw new Error("VIDEO_FRAME_CAPTURE_FAILED")

  canvas.width = 1280
  canvas.height = 720

  const frames: SampledFrame[] = []
  for (const ratio of ratios) {
    video.currentTime = Math.max(0, Math.min(video.duration * ratio, video.duration - 0.1))
    await new Promise<void>((resolve, reject) => {
      video.onseeked = () => resolve()
      video.onerror = () => reject(new Error("VIDEO_SEEK_FAILED"))
    })

    context.drawImage(video, 0, 0, canvas.width, canvas.height)
    const dataUrl = canvas.toDataURL("image/jpeg", 0.72)
    frames.push({
      dataUrl,
      ratio,
      sharpness: 0.8,
      brightness: 0.5,
      hash: `${Math.round(ratio * 1000)}`,
    })
  }

  URL.revokeObjectURL(objectUrl)
  return frames
}
```
