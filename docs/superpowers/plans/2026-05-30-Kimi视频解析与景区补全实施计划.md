# Kimi视频解析与景区补全实施计划 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为当前景点速查页接入基于项目内置测试视频的 Kimi 两阶段解析链路，实现“视频理解 + 景区联网补全”，并将最终结果渲染为现有 `SpotGuide` 页面。

**Architecture:** 继续保持前端单页架构，在前端新增测试视频入口、JSON 请求模板读取、Kimi 两阶段调用 service、第一阶段结果到 `SpotGuide` 的转换层，以及降级状态处理。默认读取项目内置测试视频，第二阶段使用 Kimi `$web_search` 补全信息，失败时降级展示第一阶段基础结果。

**Tech Stack:** React 18、TypeScript、Vite、Tailwind CSS、Zustand、Vitest、Kimi OpenAI-Compatible API

import { MAX_DEMO_VIDEO_SIZE_MB, SUPPORTED_VIDEO_TYPES } from "@/config/demoVideo"

type ValidationResult =
  | { ok: true }
  | { ok: false; reason: "UNSUPPORTED_TYPE" | "FILE_TOO_LARGE" | "EMPTY_FILE" }

export function validateVideoFile(file: File): ValidationResult {
  if (file.size <= 0) return { ok: false, reason: "EMPTY_FILE" }
  if (!SUPPORTED_VIDEO_TYPES.includes(file.type)) return { ok: false, reason: "UNSUPPORTED_TYPE" }
  if (file.size > MAX_DEMO_VIDEO_SIZE_MB * 1024 * 1024) return { ok: false, reason: "FILE_TOO_LARGE" }
  return { ok: true }
}
```

export async function readFileAsDataUrl(file: File): Promise<string> {
  return await new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(String(reader.result))
    reader.onerror = () => reject(new Error("FILE_READ_FAILED"))
    reader.readAsDataURL(file)
  })
}
```

import type { Stage1SpotExtraction } from "@/types/spotGuide"

export async function buildVideoAnalysisRequest(videoDataUrl: string) {
  return {
    model: stage1Template.model,
    temperature: stage1Template.temperature,
    response_format: stage1Template.response_format,
    messages: [
      { role: "system", content: stage1Template.system_prompt },
      { role: "user", content: stage1Template.user_prompt_template.replace("{{VIDEO_DATA_URL}}", videoDataUrl) },
    ],
  }
}

export async function buildSpotEnrichmentRequest(stage1: Stage1SpotExtraction) {
  return {
    model: stage2Template.model,
    tools: stage2Template.tools,
    messages: [
      { role: "system", content: stage2Template.system_prompt },
      { role: "user", content: stage2Template.user_prompt_template.replace("{{STAGE1_JSON}}", JSON.stringify(stage1)) },
    ],
  }
}

import { buildSpotEnrichmentRequest, buildVideoAnalysisRequest } from "@/utils/promptTemplates"
import type { SpotGuide, Stage1SpotExtraction } from "@/types/spotGuide"

const API_BASE = "https://api.moonshot.ai/v1/chat/completions"

async function postJson(body: unknown) {
  const apiKey = (window as Window & { __KIMI_API_KEY__?: string }).__KIMI_API_KEY__
  if (!apiKey) throw new Error("MISSING_KIMI_API_KEY")

  const response = await fetch(API_BASE, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
      Authorization: `Bearer ${apiKey}`,
    },
    body: JSON.stringify(body),
  })

  if (!response.ok) throw new Error("KIMI_REQUEST_FAILED")
  return await response.json()
}

export async function analyzeVideoWithKimi(videoDataUrl: string): Promise<Stage1SpotExtraction> {
  const body = await buildVideoAnalysisRequest(videoDataUrl)
  const json = await postJson(body)
  return JSON.parse(json.choices[0].message.content) as Stage1SpotExtraction
}

export async function enrichSpotWithKimi(stage1: Stage1SpotExtraction): Promise<SpotGuide> {
  const body = await buildSpotEnrichmentRequest(stage1)
  const json = await postJson(body)
  return JSON.parse(json.choices[0].message.content) as SpotGuide
}

import { analyzeVideoWithKimi, enrichSpotWithKimi } from "@/services/kimiClient"
import { mapStage1ToSpotGuide } from "@/utils/spotGuideMapper"
import { readFileAsDataUrl, validateVideoFile } from "@/utils/videoFile"

analyzeDemoVideo: async (file) => {
  set({ status: "validating_video", error: null, partialMessage: null })
  const validation = validateVideoFile(file)
  if (!validation.ok) {
    set({ status: "error", error: { code: "PARSE_FAILED", message: validation.reason } })
    return
  }

  set({ status: "reading_video" })
  const dataUrl = await readFileAsDataUrl(file)

  set({ status: "analyzing_video" })
  const stage1 = await analyzeVideoWithKimi(dataUrl)

  try {
    set({ status: "enriching_spot" })
    const finalGuide = await enrichSpotWithKimi(stage1)
    set({ status: "success", result: finalGuide })
  } catch {
    set({
      status: "partial_success",
      result: mapStage1ToSpotGuide(stage1),
      partialMessage: "联网补充失败，当前展示的是视频提取结果",
    })
  }
}
