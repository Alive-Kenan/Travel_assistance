> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将当前均匀抽样抽帧替换为“文本倾向优先 + 场景变化优先 + 少量均匀兜底”的关键地点导向抽帧策略，以提高景区名称和关键地点命中率。

**Architecture:** 保留现有前端本地抽帧和 Kimi 多模态识别链路，只替换候选帧生成与代表帧选择逻辑。浏览器端先采更多候选帧，再通过文本倾向分、场景变化分、清晰度和去重分筛选出更可能包含牌匾、字幕、路牌和关键镜头的代表帧。若评分过程失败，则回退到均匀抽样兜底。

**Tech Stack:** TypeScript, React 18, Vite, Vitest, HTMLVideoElement, Canvas

```ts
export type GuidedFrame = SampledFrame & {
  textLikeScore: number
  sceneChangeScore: number
  guidedScore: number
}

export function buildGuidedSampleRatios(): number[] {
  return [0.08, 0.15, 0.22, 0.29, 0.36, 0.43, 0.5, 0.57, 0.64, 0.71, 0.78, 0.85]
}

export function calculateTextLikeScore(input: {
  edgeDensity: number
  contrast: number
  bottomBandDensity: number
}): number {
  return Number(
    (input.edgeDensity * 0.35 + input.contrast * 0.25 + input.bottomBandDensity * 0.4).toFixed(4),
  )
}

export function calculateSceneChangeScore(prevDiff: number, nextDiff: number): number {
  return Number(((prevDiff * 0.5) + (nextDiff * 0.5)).toFixed(4))
}

export function pickGuidedFrames(frames: GuidedFrame[], limit: number): GuidedFrame[] {
  const picked: GuidedFrame[] = []
  for (const frame of [...frames].sort((a, b) => b.guidedScore - a.guidedScore)) {
    const isNearDuplicate = picked.some((item) => hammingDistance(item.hash, frame.hash) <= 1)
    if (!isNearDuplicate) picked.push(frame)
    if (picked.length === limit) break
  }
  return picked.sort((a, b) => a.ratio - b.ratio)
}
```

```ts
export function chooseGuidedFramesWithFallback(
  frames: SampledFrame[],
  scoreFrames: () => GuidedFrame[],
  limit: number,
): SampledFrame[] {
  try {
    return pickGuidedFrames(scoreFrames(), limit)
  } catch {
    return pickRepresentativeFrames(frames, limit)
  }
}
```

```ts
export function scoreGuidedFrames(frames: SampledFrame[]): GuidedFrame[] {
  return frames.map((frame, index) => ({
    ...frame,
    textLikeScore: index % 3 === 0 ? 0.7 : 0.3,
    sceneChangeScore: index % 2 === 0 ? 0.6 : 0.4,
    guidedScore: Number((frame.sharpness * 0.2 + (index % 3 === 0 ? 0.7 : 0.3) * 0.5 + (index % 2 === 0 ? 0.6 : 0.4) * 0.3).toFixed(4)),
  }))
}
```
